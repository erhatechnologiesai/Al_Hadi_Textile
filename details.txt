4. Core Features & Code Implementations4.1 WhatsApp Direct Order FlowHar product card par ek dynamic button hoga jo user ki choice ke mutabiq text message generate karega.TypeScript// frontend/src/lib/utils.ts

export interface ProductDetails {
  id: string;
  name: string;
  slug: string;
  category: string;
  fabric: string;
  gsm: number;
  dimensions: string;
}

export function generateWhatsAppOrderUrl(product: ProductDetails, quantity: number = 1): string {
  const businessPhone = "923001234567"; // AL HADI Business Number

  const messageText = `Salam AL HADI Textiles,
Mujhe is product ka order / rate confirm karna hai:

• Product: ${product.name}
• Category: ${product.category}
• Fabric: ${product.fabric} (${product.gsm} GSM)
• Size: ${product.dimensions}
• Required Quantity: ${quantity} piece(s)

Product Link: https://alhaditextiles.com/products/${product.slug}

Meherbani karke wholesale / retail rate aur delivery details share karein.`;

  return `https://wa.me/${businessPhone}?text=${encodeURIComponent(messageText)}`;
}
4.2 Lead Logging Backend RouteCustomer jab WhatsApp button dabayega, front-end pehle backend ko lead log bhejega taake record rahe ke kis product par ziyada orders aa rahe hain:JavaScript// backend/src/controllers/inquiryController.js

const inquiries = []; // Ya database model mein save karein

export const recordInquiry = async (req, res) => {
  try {
    const { productId, productName, customerDevice, quantity } = req.body;
    
    const newInquiry = {
      id: Date.now().toString(),
      productId,
      productName,
      customerDevice: customerDevice || 'web',
      quantity: quantity || 1,
      createdAt: new Date().toISOString()
    };

    inquiries.push(newInquiry);
    
    return res.status(201).json({
      success: true,
      message: "Lead recorded successfully",
      data: newInquiry
    });
  } catch (error) {
    return res.status(500).json({ success: false, message: error.message });
  }
};
